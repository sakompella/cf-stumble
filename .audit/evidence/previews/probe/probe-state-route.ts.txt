import { Supervisor } from "./supervisor/supervisor.js";

export { Supervisor };

type ProbeEnv = { SUPERVISOR: DurableObjectNamespace<Supervisor> };

export default {
  async fetch(request: Request, env: ProbeEnv): Promise<Response> {
    const supervisor = env.SUPERVISOR.getByName("facet-spike");

    if (new URL(request.url).pathname === "/probe/state") {
      const active = await supervisor.getActiveGeneration();
      const attempts = await supervisor.getRelayAttempts();
      return Response.json({
        activeLabel: active.generation?.label ?? null,
        activeStatus: active.generation?.status ?? null,
        harnessCommit: active.generation?.harnessCommit ?? null,
        attemptCount: attempts.length,
      });
    }

    return supervisor.fetch(request);
  },
};
